Why Stack?

Stack follows LIFO. It temporarily stores operators during conversion and operands/intermediate results during evaluation.

INFIX → POSTFIX: Operator Stack SUFFIX EVALUATION: Operand / Result Stack

Infix → Postfix

Precedence: ^ > *,/,% > +,-

Algorithm

  1. Initialize stack.
  2. Scan left to right.
  3. Operand → output.
  4. ( → push.
  5. ) → pop until (.
  6. Operator → pop higher/equal, then push.
  7. End → pop remaining.

Flowchart

START ↓ Initialize ↓ Read symbol ↓ End? YES → Pop remaining → Output → STOP NO → Operand? YES → Output NO → ( ? Push ) ? Pop until ( Operator → Pop higher/equal → Push ↓ Next symbol

Suffix / Postfix Evaluation

START ↓ Initialize stack ↓ Read symbol ↓ Digit? YES → PUSH NO → POP operand2 → POP operand1 Calculate a op b → PUSH result ↓ End? YES → Pop final result → Display → STOP

Easy C Programs

Infix to Postfix

#include <stdio.h> #include <ctype.h> #define MAX 100 char stack[MAX]; int top=-1; void push(char ch){stack[++top]=ch;} char pop(){return stack[top--];} int precedence(char ch){if(ch=='^')return 3;if(ch=='*'||ch=='/'||ch=='%')return 2;if(ch=='+'||ch=='-')return 1;return 0;} void convert(char in[],char out[]){int i=0,j=0;char ch;while(in[i]!='\0'){ch=in[i];if(isalnum(ch))out[j++]=ch;else if(ch=='(')push(ch);else if(ch==')'){while(top!=-1&&stack[top]!='(')out[j++]=pop();if(top!=-1)pop();}else{while(top!=-1&&stack[top]!='('&&precedence(stack[top])>=precedence(ch))out[j++]=pop();push(ch);}i++;}while(top!=-1)out[j++]=pop();out[j]='\0';} int main(){char in[MAX],out[MAX];printf("Enter Infix Expression: ");scanf("%s",in);convert(in,out);printf("Postfix Expression: %s\n",out);return 0;}

Suffix Evaluation

#include <stdio.h> #include <ctype.h> #include <math.h> #define MAX 100 int stack[MAX],top=-1; void push(int x){stack[++top]=x;} int pop(){return stack[top--];} int calc(int a,int b,char op){switch(op){case '+':return a+b;case '-':return a-b;case '*':return a*b;case '/':return a/b;case '%':return a%b;case '^':return (int)pow(a,b);}return 0;} int eval(char e[]){int i,a,b,r;for(i=0;e[i]!='\0';i++){if(isdigit(e[i]))push(e[i]-'0');else{b=pop();a=pop();r=calc(a,b,e[i]);push(r);}}return pop();} int main(){char e[MAX];printf("Enter Suffix Expression: ");scanf("%s",e);printf("Result = %d\n",eval(e));return 0;}
